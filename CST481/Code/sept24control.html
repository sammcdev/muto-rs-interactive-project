import pygame, sys
import pygame.gfxdraw
import random
from datetime import datetime
from MutoLib import Muto
from pygame.locals import QUIT

#Initialize Pygame
pygame.init()
screen = pygame.display.set_mode((400, 400))
screen.fill('white')
pygame.display.set_caption('Hexapod Muto RS Control  Center')

clock = pygame.time.Clock()
background = pygame.Surface((400,400))
font = pygame.font.SysFont('arial',30)

#Function for setting up connection with the Hexapod Muto RS robot
def findUSB():
    for i  in range(0,5):
        try:
            print(f'ttyUSB{i}')
            g_bot = Muto(f'/dev/ttyUSB{i}')
        except:
            print(i)
            continue
        else:
            print('success')
            return i
    return -1

#Utility, unused currently
def timer(seconds):
    t1 = datetime.now()
    t1 = int(t1.second)
    t2 = datetime.now()
    t2 = int(t2.second)
    while (t2 - t1) < seconds:

#Initialize Muto robot
g_bot = Muto(f'/dev/ttyUSB{findUSB()}')
gSpeed = 1

#Runtime
while True:
    #Closing the window will end the program
    for event in pygame.event.get():
        if event.type == QUIT:
            pygame.quit()
            sys.exit()
    #Get controls
    keys = pygame.key.get_pressed()

    usedKeys = [keys[pygame.K_w],keys[pygame.K_a],keys[pygame.K_s],
                keys[pygame.K_d],keys[pygame.K_f], keys[pygame.K_q],
                keys[pygame.K_e], keys[pygame.K_i], keys[pygame.K_o],
                keys[pygame.K_p]]
    #Act based on controls
    if usedKeys[0]:
        g_bot.forward(10)
        
    elif usedKeys[1]:
        g_bot.left(10)
        
    elif usedKeys[2]:
        g_bot.back(10)
        
    elif usedKeys[3]:
        g_bot.right(10)

    elif usedKeys[4]:
        g_bot.buzzer(255)
        
    elif usedKeys[5]:
        g_bot.turnleft(10)

    elif usedKeys[6]:
        g_bot.turnright(10)

    elif usedKeys[7]:
        if gSpeed > 1:
            gSpeed -= 1

    elif usedKeys[8]:
        gSpeed = 1

    elif usedKeys[9]:
        if gSpeed < 5:
            gSpeed += 1
      
    else: #If no input, disable buzzer and reset movement
        g_bot.buzzer(0)
        g_bot.stop()
    g_bot.speed(gSpeed) #Update speed to gSpeed

    pygame.display.update() #Update Pygame display
    clock.tick(20) #Framerate of window and input detection, can be changed

import socket, struct, io, ssl
from PIL import Image
import datetime

#Send message to all
def sendToAll(clients, msg):
    for i in range(len(clients)):
        clients[i][1].sendall(msg.encode())

# Sample code provided by instructor for receiving image
def recvall(sock, n: int) -> bytes:
    """Receive exactly n bytes or raise ConnectionError."""
    data = bytearray()
    while len(data) < n:
        chunk = sock.recv(n - len(data))
        if not chunk:
            raise ConnectionError("Socket connection closed early")
        data += chunk
    return bytes(data)

def handle_client(conn, addr, errors, file = "PNG"):
    name = ''
    try:
        # Read 8-byte big-endian length
        header = recvall(conn, 8)
        (length,) = struct.unpack(">Q", header)
        # Read image payload
        payload = recvall(conn, length)

        # Turn bytes -> PIL.Image
        img = Image.open(io.BytesIO(payload)).convert("RGB")

        # Get time for unique name
        time = datetime.datetime.now()
        time = str(time)
        time = time.split(".")
        time = time[0] + "-" + time[1]
        time = time.split(":")
        newTime = ""
        for i in time:
            newTime += i
        newTime = newTime.split(" ")
        newTime = newTime[0] + "-" + newTime[1]
        name = newTime
        
        #print(f"Received image {img.size} mode={img.mode} from {addr}")
        # Do something with it: save or process
        extension = 'jpg' if file == 'JPEG' else 'png'
        img.save(f"snapshot.{extension}", format = file)
        errors = 0
    except Exception as e:
        #print("Error:", e)
        errors += 1
        # If more than 10 errors, end 
        if errors > 10:
            print('Closing connection.')
            return True, errors, ''
    
    return False, errors, name

def pil_to_bytes(img: Image.Image, fmt="JPEG", q = 70) -> bytes:
    # Encode a PIL image to bytes (PNG/JPEG/etc.)
    buf = io.BytesIO()
    img.save(buf, format=fmt, quality = q)
    return buf.getvalue()

def main():
    # Create a server socket
    context = ssl.create_default_context(ssl.Purpose.CLIENT_AUTH)
    context.load_cert_chain(certfile='cert.pem', keyfile='cert.pem')

    server_socket = socket.socket(socket.AF_INET,socket.SOCK_STREAM)
    server_socket.bind(('0.0.0.0', 5050)) # Bind IP and port
    server_socket.listen()
    print("Server is running... Waiting for the clients to connect.")
    # Accept a client connection
    clients = []
    count = 0
    while count < 2:
        client_socket, addr = server_socket.accept()
        client_socket = context.wrap_socket(client_socket, server_side=True)
        rsp = client_socket.recv(1024).decode()
        clients.append([rsp, client_socket, addr])
        print(clients[count])
        count += 1

    mutoIndex, jetsonIndex = 0, 0
    count = 0

            count += 1

    print('muto:', mutoIndex)

        jetsonEnd = clients[jetsonIndex][1].recv(1024).decode()
        if jetsonEnd == 'END':

main()

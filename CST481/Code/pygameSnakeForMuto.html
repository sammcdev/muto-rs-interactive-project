#Search for elements on the game board
def searchSnakeHead(gameBoard):
    for row in range(len(gameBoard)):
        for col in range(len(gameBoard)):
            if gameBoard[row][col] == 1:
                #print("Snake Head: ",row, col)
                return row, col
            
def searchSnakeTrail(gameBoard):
    trail = []
    for row in range(len(gameBoard)):
        for col in range(len(gameBoard)):
            if gameBoard[row][col] == 0.5:
                trail.append([row, col])
    return trail

def searchApple(gameBoard):
    #print("finding apple here ", gameBoard)
    for row in range(len(gameBoard)):
        for col in range(len(gameBoard)):
            #print(gameBoard[row][col])
            if gameBoard[row][col] == 2:
                #print('apple found')
                return row, col
            
#Check to see if the apple is in a certain coordinate
def checkApple(row, col, apRow, apCol):
    if row == apRow and apCol == col:
        return True
    return False

#Movement
def up(gameBoard, trail):
    #Get rows and cols of snake head and apple
    row, col = searchSnakeHead(gameBoard)
    apRow, apCol = searchApple(gameBoard)
    score = 0
    #Move the snake head
    gameBoard[row][col] = 0
    gameBoard[row-1][col] = 1
    #Add previous location to trail
    trail.append([row, col])
    #Check to see if the head is now on the apple, and increment score if so
    nR, nC = searchSnakeHead(gameBoard)
    if checkApple(nR, nC, apRow, apCol):
        score += 1
        randomApple(gameBoard)
    return score

def down(gameBoard, trail):
    #Get rows and cols of snake head and apple
    row, col = searchSnakeHead(gameBoard)
    apRow, apCol = searchApple(gameBoard)
    score = 0
    #Move the snake head, make sure it doesn't leave the bounds of the array
    if row+1 < 5:
        gameBoard[row][col] = 0
        gameBoard[row+1][col] = 1
        trail.append([row, col])
    else:
        gameBoard[row][col] = 0
        gameBoard[0][col] = 1
        trail.append([row, col])
    #Check to see if the head is now on the apple, and increment score if so
    nR, nC = searchSnakeHead(gameBoard)
    if checkApple(nR, nC, apRow, apCol):
        score += 1
        randomApple(gameBoard)
    return score

def left(gameBoard, trail):
    #Get rows and cols of snake head and apple
    row, col = searchSnakeHead(gameBoard)

    score = 0
    #Move the snake head

    return score

def right(gameBoard, trail):
    #Get rows and cols of snake head and apple
    row, col = searchSnakeHead(gameBoard)
    apRow, apCol = searchApple(gameBoard)
    score = 0
    #Move the snake head, make sure it doesn't leave the bounds of the array
    if col+1 < 5:
        gameBoard[row][col] = 0
        gameBoard[row][col+1] = 1
    else:
        gameBoard[row][col] = 0
        gameBoard[row][0] = 1
    trail.append([row, col])
    #Check to see if the head is now on the apple, and increment score if so
    nR, nC = searchSnakeHead(gameBoard)
    if checkApple(nR, nC, apRow, apCol):
        score += 1
        randomApple(gameBoard)
    return score

#Handle trail creation and deletion
def delTrail(gameBoard, trail, trailLegacy, score):
    #Get rid of excess
    while len(trail) > score:
        trail.pop(0)
    while len(trailLegacy) > score+1:
        trailLegacy.pop(0)
    trailR = []
    #Reverse so it is in the correct order of construction
    for i in trail:
        trailR.append(i)
    trailR.reverse()
    apRow, apCol = searchApple(gameBoard)
    row, col = searchSnakeHead(gameBoard)
    s = score
    # Make sure the tail length is set to a max
    if score > 5:
        s = 5
    #Draw the tail
    for i in range(s):
        trailRow = trailR[i][0]
        trailCol = trailR[i][1] 
        if gameBoard[trailRow][trailCol] == 0:
            gameBoard[trailRow][trailCol] = 0.5
    dispTrail = searchSnakeTrail(gameBoard)
    removeTrail = []
    #Remove the extra tail segments
    for i in dispTrail:
        if i not in trail:
            removeTrail.append(i)
    for r in range(len(gameBoard)):
        for c in range(len(gameBoard)):
            if gameBoard[r][c] == 0.5 and [r,c] in removeTrail:
                gameBoard[r][c] = 0

#Spawn a new apple
def randomApple(gameBoard):
    while True:
        appleX = random.randint(0,4)
        appleY = random.randint(0,4)
        if gameBoard[appleX][appleY] == 0:
            gameBoard[appleX][appleY] = 2
            break
        
#Checks for game end state and signals it when true
def checkGameOver(gameBoard, trail):
    row, col = searchSnakeHead(gameBoard)
    #print(trail)
    #print(row,col)
    if [row, col] in trail:
        print('Game Over')
        return True
    return False

def timer(seconds):
    t1 = datetime.now()
    t1 = int(t1.second)
    t2 = datetime.now()
    t2 = int(t2.second)
    while (t2 - t1) < seconds:
        t2 = datetime.now()
        t2 = int(t2.second)

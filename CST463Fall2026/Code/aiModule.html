def ask(q):
    prompt = "You are a robot named Muto. Do not say your name unless specifially asked. You must keep the response under 100 characters, so use simple language and be concise. This is what you are responding to: " + q
    content = [
        {"role" : "user",
            "content": prompt
        }
    ]
    try:
        response = generator(
            content,
            max_new_tokens=30,
            do_sample = True,
            num_return_sequences=1
        )
        reply = response[0]["generated_text"][-1]['content']
        return reply
    except Exception as e:
        return 'There was an error with generation.'

def askImage(filePath):
    try:
        # https://huggingface.co/HuggingFaceTB/SmolVLM-256M-Instruct
        image = Image.open(
            filePath
        ).convert("RGB")
        #content = imageProcessor.apply_chat_template(content, add_generation_prompt = True)
        inputs = imageProcessor(image, 
                                return_tensors="pt"
        )
        output = imageModel.generate(**inputs,  max_new_tokens=30)
        caption = imageProcessor.decode(output[0], skip_special_tokens=True)
        return caption
    except Exception as e:
        return 'There was an error with generation or processing.'

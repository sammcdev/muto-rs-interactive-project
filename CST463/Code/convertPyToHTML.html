import datetime

def main():
    path = input('Provide a file path and file name: ')
    file = 0
    try:
        file = open(path)
    except:
        print('That is not a viable file path! Try again.')
        main()
        return

    newContents = []
    for i in file:
        newLine = i
        endCharInd = i.find('\n')
        if '#' in i:
            ind = i.find('#')
            if ind == 0:
                newLine = '<span class = "comment">' + i[0:endCharInd] + '\n'
            else:
                newLine = i[0:ind] + '<span class = "comment">' + i[ind:endCharInd] + '\n'
        elif "'''" in i:
            ind = i.find("'''")
            if ind == 0:
                newLine = '<span class = "triplequote">' + i[ind:endCharInd] + '\n'
            else:
                newLine = i[0:ind] + '<span class = "triplequote">' + i[ind:endCharInd] + '\n'
        elif '"""' in i:
            ind = i.find('"""')
            if ind == 0:
                newLine = '<span class = "triplequote">' + i[ind:endCharInd] + '\n'
            else:
                newLine = i[0:ind] + '<span class = "triplequote">' + i[ind:endCharInd] + '\n'
        elif 'import' in i:
            if "'" not in i and '''"''' not in i and "'''" not in i and '#' not in i:
                newLine = '<span class = "import">' + i[0:endCharInd] + '\n'
        if '<' in i:
            ind = i.find('<')
            newLine = i[:ind] + '<' + i[ind+1:]
        newContents.append(newLine)
        
    newString = ''
    for i in newContents:
        newString += i
        
    time = datetime.datetime.now()
    time = str(time)
    time = time.split(".")
    time = time[0] + "-" + time[1]
    time = time.split(":")
    newTime = ""
    for i in time:
        newTime += i
    newTime = newTime.split(" ")
    newTime = newTime[0] + "-" + newTime[1]
        
    newFile = open(f'{newTime}.txt', 'w')
    newFile.write(newString)
    
main()

joysticks = {}

background = pygame.Surface((400,400))
font = pygame.font.SysFont('arial',30)

def timer(seconds):
    t1 = datetime.now()
    t1 = int(t1.second)
    t2 = datetime.now()
    t2 = int(t2.second)
    while (t2 - t1) < seconds:
        t2 = datetime.now()
        t2 = int(t2.second)

def wave(g_bot):
    for i in range(3):
        g_bot.motor(2, -90, 100)
        g_bot.motor(3, 90, 100)
        time.sleep(0.3)
        g_bot.stop()
        g_bot.reset()
        time.sleep(0.3)

def dance(g_bot):
    shiftDown = 7
    shiftUp = 4
    for i in range(3):
        g_bot.motor(2, shiftDown, 100)
        g_bot.motor(5, shiftDown, 100)
        g_bot.motor(8, shiftDown, 100)

        g_bot.motor(12, shiftUp, 100)
        g_bot.motor(15, shiftUp, 100)
        g_bot.motor(18, shiftUp, 100)

        time.sleep(0.75)
        g_bot.stop()
        g_bot.reset()
        time.sleep(0.5)

        g_bot.motor(11, shiftDown, 100)
        g_bot.motor(14, shiftDown, 100)
        g_bot.motor(17, shiftDown, 100)

        g_bot.motor(3, shiftUp, 100)
        g_bot.motor(6, shiftUp, 100)
        g_bot.motor(9, shiftUp, 100)
        
        time.sleep(0.75)
        g_bot.stop()
        g_bot.reset()
        time.sleep(0.5)

def peacockSpider(g_bot):
    g_bot.motor(4, 45, 100)
    g_bot.motor(13, -45, 100)
    time.sleep(0.25)

    for i in range(3):
        g_bot.motor(2, -90, 200)
        g_bot.motor(3, 90, 200)
        g_bot.motor(17, -90, 200)
        g_bot.motor(18, 90, 200)
        time.sleep(0.5)
        g_bot.motor(2, 0, 200)
        g_bot.motor(3, 0, 200)
        g_bot.motor(17, 0, 200)
        g_bot.motor(18, 0, 200)
        time.sleep(0.3)
    
    g_bot.stop()
    g_bot.reset()

# Initialize Muto robot
g_bot = Muto(f'/dev/ttyUSB{findUSB()}')
gSpeed = 1
gStep = 10
headHeight = 5
emoteChoice = 0

# Runtime
while True:
    # Closing the window will end the program
    for event in pygame.event.get():
        if event.type == QUIT:
            pygame.quit()
            sys.exit()
        elif event.type == pygame.WINDOWFOCUSLOST:
            screen = pygame.display.set_mode((400, 400))
                
            # Handle hotplugging
        if event.type == pygame.JOYDEVICEADDED:
                # This event will be generated when the program starts for every
                # joystick, filling up the list without needing to create them manually.
            joy = pygame.joystick.Joystick(event.device_index)
            joysticks[joy.get_instance_id()] = joy
            print(f"Joystick {joy.get_instance_id()} connencted")

        if event.type == pygame.JOYDEVICEREMOVED:
            del joysticks[event.instance_id]
            print(f"Joystick {event.instance_id} disconnected")
    # Get controls
    axes, buttons, hats = 0,0,0
    for joystick in joysticks.values():
        axes = joystick.get_numaxes()
        buttons = joystick.get_numbuttons()
        hats = joystick.get_numhats()

    hat = joystick.get_hat(0)
        
    name = joystick.get_name()
    
    leftX = joystick.get_axis(0)
    leftY = joystick.get_axis(1)
    
    keys = pygame.key.get_pressed()

    usedKeys = [keys[pygame.K_w],keys[pygame.K_a],keys[pygame.K_s],
                keys[pygame.K_d],keys[pygame.K_f], keys[pygame.K_q],
                keys[pygame.K_e], keys[pygame.K_i], keys[pygame.K_o],
                keys[pygame.K_p], keys[pygame.K_r], keys[pygame.K_g],
                keys[pygame.K_v], keys[pygame.K_z], 
                keys[pygame.K_j], keys[pygame.K_k], keys[pygame.K_l]]
    # Act based on controls
    # On the controller used, down is positive
    downT, upT = 0.5, -0.5 # Full down is 1 and full up is -1 on the controller used
    leftT, rightT = -0.5, 0.5
    if usedKeys[0] or hat[1] == 1 or (leftX > leftT and leftX < rightT and leftY < upT):
        g_bot.forward(gStep)
        
    elif usedKeys[1] or hat[0] == -1 or (leftX < leftT and leftY > upT and leftY < downT):
        g_bot.left(gStep)
        
    elif usedKeys[2] or hat[1] == -1 or (leftX > leftT and leftX < rightT and leftY > downT):
        g_bot.back(gStep)
        
    elif usedKeys[3] or hat[0] == 1 or (leftX > rightT and leftY > upT and leftY < downT):
        g_bot.right(gStep)
        
    elif usedKeys[5] or joystick.get_button(6):
        g_bot.turnleft(gStep)

    elif usedKeys[6] or joystick.get_button(7):
        g_bot.turnright(gStep)

    elif usedKeys[7] or joystick.get_button(8):
        if gSpeed > 1:
            gSpeed -= 1

    elif usedKeys[15] or joystick.get_button(3): # X 
        dance(g_bot)

    elif usedKeys[9] or joystick.get_button(9):
        if gSpeed < 5:
            gSpeed += 1

    elif usedKeys[10] or joystick.get_button(10):
        if gStep > 10:
            gSpeed -= 1

    elif usedKeys[12] or joystick.get_button(11):
        if gStep < 25:
            gSpeed += 1

    elif usedKeys[11]: # G on keyboard
        gStep = 10

    elif usedKeys[8]: # O on keyboard, reset speed
        gSpeed = 1

    elif usedKeys[14] or joystick.get_button(1): # B
        wave(g_bot)
        
    elif usedKeys[16] or joystick.get_button(4): # Y on controller
        peacockSpider(g_bot)
      
    else: # If no input, disable buzzer and reset movement
        g_bot.stop()

    if usedKeys[4] or joystick.get_button(0): # A    
        g_bot.buzzer(255)
    else:
        g_bot.buzzer(0)
        
    g_bot.speed(gSpeed) # Update speed to gSpeed

    pygame.display.update() # Update Pygame display
    clock.tick(20) # Framerate of window and input detection, can be changed
